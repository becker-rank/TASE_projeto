from machine import Pin, PWM

import threading

# 1. Create the Thread objects
# Note: args must be a tuple, so use a trailing comma for a single argument
t1 = threading.Thread(target=transmission.trans, args=("report.pdf", 2))
t2 = threading.Thread(target=reception.rec, args=("data.csv", 3))

# 2. Start the threads (they run concurrently)
t1.start()
t2.start()

# 3. Wait for both threads to finish before continuing
t1.join()
t2.join()
